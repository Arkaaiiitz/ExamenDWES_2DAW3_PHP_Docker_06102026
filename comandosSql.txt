/*CREACIÓN DE LA BASE DE DATOS CAE*/
CREATE DATABASE IF NOT EXISTS CAE;

/*SELECCIÓN DE LA BASE DE DATOS PARA TRABAJAR SOBRE ELLA*/
USE CAE;

/*CREACIÓN DE LA TABLA SOLICITUD*/
CREATE TABLE IF NOT EXISTS SOLICITUD (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    dni varchar(9) NOT NULL,
    f_nac DATE NOT NULL,
    tlf VARCHAR(9) NOT NULL,
    email VARCHAR(100) NOT NULL,
    profesion VARCHAR(50) NOT NULL,
    jornadaParcial BOOLEAN NOT NULL,
    idiomas VARCHAR(150)
)
